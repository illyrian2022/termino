<script>
function rentalCleanUI() {
    return {
        vehicle: '',
        price: 0,
        pickupDate: '',
        returnDate: '',
        pickupTime: '10:00',
        returnTime: '10:00',
        step: 'pickup',
        busy: [],

        cursor: new Date(
            new Date().getFullYear(),
            new Date().getMonth(),
            1
        ),

        async init() {
            const first = @json($rentalVehicles->first());

            if (first) {
                this.vehicle = String(first.id);
                this.price = Number(first.price_per_day);
                await this.loadAvailability();
            }
        },

        async selectVehicle(id, price) {
            this.vehicle = String(id);
            this.price = Number(price);
            this.resetDates();
            await this.loadAvailability();
        },

        resetDates() {
            this.pickupDate = '';
            this.returnDate = '';
            this.step = 'pickup';
        },

        iso(day) {
            const y = this.cursor.getFullYear();
            const m = String(this.cursor.getMonth()+1).padStart(2,'0');
            return `${y}-${m}-${String(day).padStart(2,'0')}`;
        },

        async loadAvailability() {
            if (!this.vehicle) return;

            const from = this.iso(1);

            const end = new Date(
                this.cursor.getFullYear(),
                this.cursor.getMonth()+2,
                0
            );

            const to =
                `${end.getFullYear()}-` +
                `${String(end.getMonth()+1).padStart(2,'0')}-` +
                `${String(end.getDate()).padStart(2,'0')}`;

            try {
                const url = new URL(
                    @json(route('rental.public.availability', $business)),
                    location.origin
                );

                url.searchParams.set('vehicle_id', this.vehicle);
                url.searchParams.set('from', from);
                url.searchParams.set('to', to);

                const response = await fetch(url, {
                    headers:{Accept:'application/json'},
                    cache:'no-store'
                });

                const data = await response.json();
                this.busy = data.busy || [];

            } catch(e) {
                this.busy = [];
            }
        },

        monthTitle() {
            return this.cursor.toLocaleDateString(
                document.documentElement.lang || 'sq',
                {month:'long',year:'numeric'}
            );
        },

        monthDays() {
            const count = new Date(
                this.cursor.getFullYear(),
                this.cursor.getMonth()+1,
                0
            ).getDate();

            return Array.from({length:count},(_,i)=>i+1);
        },

        blanks() {
            let d = new Date(
                this.cursor.getFullYear(),
                this.cursor.getMonth(),
                1
            ).getDay();

            d = d === 0 ? 7 : d;
            return Array.from({length:d-1},(_,i)=>i);
        },

        isBusy(day) {
            return this.busy.includes(this.iso(day));
        },

        isPast(day) {
            const d = new Date(`${this.iso(day)}T12:00:00`);
            const today = new Date();
            today.setHours(0,0,0,0);
            return d < today;
        },

        disabled(day) {
            return this.isBusy(day) || this.isPast(day);
        },

        chooseDate(day) {
            if (this.disabled(day)) return;

            const value = this.iso(day);

            if (this.step === 'pickup') {
                this.pickupDate = value;
                this.returnDate = '';
                this.step = 'return';
                return;
            }

            if (value <= this.pickupDate) {
                this.pickupDate = value;
                this.returnDate = '';
                this.step = 'return';
                return;
            }

            if (this.busy.some(d => d >= this.pickupDate && d <= value)) {
                return;
            }

            this.returnDate = value;
            this.step = 'pickup';
        },

        dayClass(day) {
            const value = this.iso(day);

            if (this.isPast(day))
                return 'text-slate-800 cursor-not-allowed';

            if (this.isBusy(day))
                return 'text-rose-400/40 cursor-not-allowed';

            if (value === this.pickupDate || value === this.returnDate)
                return 'bg-cyan-400 text-[#07111d] shadow-lg shadow-cyan-950/30';

            if (
                this.pickupDate &&
                this.returnDate &&
                value > this.pickupDate &&
                value < this.returnDate
            )
                return 'bg-cyan-400/10 text-cyan-300';

            return 'text-slate-300 hover:bg-white/[.05]';
        },

        pretty(value) {
            if (!value) return '';

            return new Date(`${value}T12:00:00`)
                .toLocaleDateString(
                    document.documentElement.lang || 'sq',
                    {day:'2-digit',month:'short',year:'numeric'}
                );
        },

        days() {
            if (!this.pickupDate || !this.returnDate) return 0;

            const a = new Date(`${this.pickupDate}T00:00:00`);
            const b = new Date(`${this.returnDate}T00:00:00`);

            return Math.max(1,Math.ceil((b-a)/86400000));
        },

        total() {
            return (this.days()*this.price).toFixed(0);
        },

        pickupValue() {
            return this.pickupDate
                ? `${this.pickupDate}T${this.pickupTime}`
                : '';
        },

        returnValue() {
            return this.returnDate
                ? `${this.returnDate}T${this.returnTime}`
                : '';
        },

        async prevMonth() {
            const now = new Date();

            const candidate = new Date(
                this.cursor.getFullYear(),
                this.cursor.getMonth()-1,
                1
            );

            const min = new Date(
                now.getFullYear(),
                now.getMonth(),
                1
            );

            if (candidate < min) return;

            this.cursor = candidate;
            await this.loadAvailability();
        },

        async nextMonth() {
            this.cursor = new Date(
                this.cursor.getFullYear(),
                this.cursor.getMonth()+1,
                1
            );

            await this.loadAvailability();
        }
    }
}
</script>
